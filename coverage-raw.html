package main

import (
        "context"
        "log/slog"
        "os"
        "os/signal"
        "syscall"
        "time"

        "github.com/saxeum/godough/internal/app"
        "github.com/saxeum/godough/internal/config"
)

// bootTimeout bounds how long startup may spend reaching the database. A DSN
// that points at a host which drops packets would otherwise block for the
// operating system's TCP timeout, with the pod neither started nor crashed.
const bootTimeout = 10 * time.Second

func main() {
        logger := slog.New(slog.NewTextHandler(os.Stdout, nil))

        // The signal context is built here and passed in, so the shutdown path can be
        // driven by a test that cancels a context instead of signalling the test
        // process. Sending a real signal to your own process is a test that can take
        // the whole suite with it.
        ctx, stop := signal.NotifyContext(context.Background(), syscall.SIGINT, syscall.SIGTERM)
        defer stop()

        // run owns every resource, so its defers run on the way out.
        if err := run(ctx, logger); err != nil {
                logger.Error("fatal", "err", err)
                os.Exit(1)
        }
}

// run is everything main does once the signals are wired: read the environment,
// reach the database, serve until ctx ends, and release. Its defers matter — the
// pool is closed on the way out of here, after the server has drained.
func run(ctx context.Context, logger *slog.Logger) error {
        cfg, err := config.FromEnv()
        if err != nil {
                return err
        }

        bootCtx, bootDone := context.WithTimeout(ctx, bootTimeout)
        application, err := app.New(bootCtx, cfg, logger)
        bootDone()
        if err != nil {
                return err
        }
        defer application.Close()

        logger.Info("starting server", "addr", cfg.Addr)
        return application.Run(ctx)
}

const (
        readHeaderTimeout = 5 * time.Second
        // readTimeout bounds the whole request, headers and body. The body is capped
        // at 1 MiB in the transport, so a client that needs longer than this is
        // trickling bytes on purpose.
        readTimeout = 10 * time.Second
        // writeTimeout must exceed the handler's own deadline, or the connection is
        // cut before the handler can report why it gave up.
        writeTimeout = httpapi.RequestTimeout + 5*time.Second
        idleTimeout  = 60 * time.Second
        // maxHeaderBytes is 1 MiB, the same as the body cap. Go's default is 1 MiB
        // too, but naming it means a change here is visible rather than inherited.
        maxHeaderBytes = 1 << 20

        // shutdownGrace is how long a draining server waits for in-flight requests.
        // Kubernetes' terminationGracePeriodSeconds must exceed this plus its preStop
        // hook, or the pod is killed mid-drain. It is a constant and not a setting
        // on purpose: the relationship to the manifest is the thing that matters, and
        // two places to configure one number is how they come to disagree.
        shutdownGrace = 10 * time.Second
)

// newHTTPServer builds the server with every limit named.
//
// It is separate from New so the limits can be asserted without a database, and
// so that "which timeout is this" is answered by a constant rather than by a
// literal inside a struct.
func newHTTPServer(addr string, handler http.Handler, log *slog.Logger) *http.Server {
        return &http.Server{
                Addr:              addr,
                Handler:           handler,
                ReadHeaderTimeout: readHeaderTimeout,
                ReadTimeout:       readTimeout,
                WriteTimeout:      writeTimeout,
                IdleTimeout:       idleTimeout,
                MaxHeaderBytes:    maxHeaderBytes,
                // Everything Go's server wants to say — TLS handshakes, handler errors,
                // panics that escaped a handler — goes through the application's logger
                // instead of the standard library's default one.
                ErrorLog: slog.NewLogLogger(log.Handler(), slog.LevelError),
        }
}

// App is a running configuration: the database pool, the HTTP server built over
// the layers, and the logger they share.
type App struct {
        log *slog.Logger
        db  *pgxpool.Pool
        srv *http.Server
}

// New opens the database, proves it is reachable, and builds the chain
// repository -> service -> transport. A failure here is a boot failure.
//
// ctx carries the boot deadline. It is the caller's, not a constant in here, so
// that a database which accepts the connection and never answers fails the
// process in bounded time instead of parking a pod in a half-started state.
func New(ctx context.Context, cfg config.Config, log *slog.Logger) (*App, error) {
        db, err := postgres.NewPool(ctx, cfg.DatabaseURL)
        if err != nil {
                return nil, err
        }

        if err := db.Ping(ctx); err != nil {
                db.Close()
                return nil, fmt.Errorf("ping database: %w", err)
        }

        repo := postgres.NewSnippetRepository(db)
        svc := service.NewSnippetService(repo)
        handler := httpapi.NewServer(log, svc, db.Ping).Routes()

        return &App{
                log: log,
                db:  db,
                srv: newHTTPServer(cfg.Addr, handler, log),
        }, nil
}

// Run binds the configured address and serves until ctx is cancelled, then
// drains. It returns the error that stopped it, or nil after a clean shutdown.
//
// Binding before serving means a taken port is an error the caller sees, not a
// failure inside a goroutine racing the rest of startup.
func (a *App) Run(ctx context.Context) error {
        listener, err := net.Listen("tcp", a.srv.Addr)
        if err != nil {
                return fmt.Errorf("listen on %s: %w", a.srv.Addr, err)
        }

        return a.Serve(ctx, listener)
}

// Serve serves an already-bound listener until ctx is cancelled, then drains.
//
// Split from Run so a test can bind port 0, learn the address, and cancel the
// context — with the listen call buried inside Run there was nothing to drive.
func (a *App) Serve(ctx context.Context, listener net.Listener) error {
        serveErr := make(chan error, 1)
        go func() {
                err := a.srv.Serve(listener)
                if errors.Is(err, http.ErrServerClosed) {
                        a.log.Info("stopped serving new connections")
                        return
                }
                serveErr <- fmt.Errorf("serve: %w", err)
        }()

        select {
        case err := <-serveErr:
                return err
        case <-ctx.Done():
        }

        shutdownCtx, release := context.WithTimeout(context.Background(), shutdownGrace)
        defer release()

        if err := a.srv.Shutdown(shutdownCtx); err != nil {
                // Shutdown returning an error means requests are still running. Close the
                // connections so they stop, rather than letting the deferred pool Close
                // pull the database out from under handlers that are mid-query.
                a.srv.Close()
                return fmt.Errorf("shutdown: %w", err)
        }
        a.log.Info("graceful shutdown complete")
        return nil
}

// Close releases the database pool. It is separate from Run so the caller
// controls the order: the server drains first, then the pool goes.
func (a *App) Close() {
        a.db.Close()
}

// FromEnv reads and validates the environment.
func FromEnv() (Config, error) {
        addr := os.Getenv("SNIPPET_ADDR")
        if addr == "" {
                return Config{}, errors.New("SNIPPET_ADDR is not set")
        }

        dsn := os.Getenv("DATABASE_URL")
        if dsn == "" {
                return Config{}, errors.New("DATABASE_URL is not set")
        }

        return Config{Addr: addr, DatabaseURL: dsn}, nil
}

        Expires time.Time

// ValidationError carries every rejected field at once, so a form can show them
// all rather than one per round trip. The names are the JSON names, not the Go
// ones: this type is serialised directly.
type ValidationError struct {
        Fields map[string]string
}

func (e *ValidationError) Error() string {
        keys := make([]string, 0, len(e.Fields))
        for k := range e.Fields {
                keys = append(keys, k)
        }
        sort.Strings(keys)
        return fmt.Sprintf("invalid fields: %v", keys)
}

import (

// Session limits applied to every connection in the pool.
//
// Postgres defaults statement_timeout to zero, which is no limit at all, so a
// query that waits on a lock or scans a growing table holds its connection until
// the client gives up. With one connection per concurrent request, a handful of
// those is the whole replica. A bound here is the difference between a slow
// query and an outage.
//
// statement_timeout must stay below any per-request deadline (#36): if the
// client gives up first, the server keeps working on a request nobody wants.
const (
        StatementTimeout = "5s"
        LockTimeout      = "2s"
)

// NewPool opens the connection pool and applies the session limits to every
// connection, including ones opened later to replace a dropped connection —
// which is why this is AfterConnect and not three statements after startup.
func NewPool(ctx context.Context, dsn string) (*pgxpool.Pool, error) {
        cfg, err := pgxpool.ParseConfig(dsn)
        if err != nil {
                return nil, fmt.Errorf("parse database url: %w", err)
        }

        cfg.AfterConnect = func(ctx context.Context, conn *pgx.Conn) error {
                // set_config rather than SET: SET does not take parameters.
                if _, err := conn.Exec(ctx, `SELECT set_config('statement_timeout', $1, false)`, StatementTimeout); err != nil {
                        return fmt.Errorf("set statement_timeout: %w", err)
                }
                if _, err := conn.Exec(ctx, `SELECT set_config('lock_timeout', $1, false)`, LockTimeout); err != nil {
                        return fmt.Errorf("set lock_timeout: %w", err)
                }
                return nil
        }

        pool, err := pgxpool.NewWithConfig(ctx, cfg)
        if err != nil {
                return nil, fmt.Errorf("open database: %w", err)
        }

        return pool, nil
}

// Package postgres is the storage layer: the only package in the tree that
// knows SQL. It implements service.SnippetRepository and translates pgx's
// vocabulary into the domain's — ErrNoRows becomes snippet.ErrNotFound, and a
// driver error keeps its cause for the log.
package postgres

import (
        "context"
        "errors"
        "fmt"
        "time"

        "github.com/jackc/pgx/v5"
        "github.com/jackc/pgx/v5/pgxpool"

        "github.com/saxeum/godough/internal/domain/snippet"
)

// row is the shape of one snippets row: the columns of the query, in order.
//
// It exists so the scan and the query are in the same file. Scanning straight
// into snippet.Snippet, matching by field name, made
// the domain package implicitly coupled to column names and turned "add a field
// to the domain type" into a runtime failure in a package the author was not
// looking at. Positional scanning into a local struct keeps the coupling here,
// next to the SQL that has to change with it.
type row struct {
        ID      int64
        Title   string
        Content string
        Created time.Time
        Expires time.Time
}

func (r row) toDomain() snippet.Snippet {
        return snippet.Snippet{
                ID:      r.ID,
                Title:   r.Title,
                Content: r.Content,
                Created: r.Created,
                Expires: r.Expires,
        }
}

// SnippetRepository talks to the snippets table.
type SnippetRepository struct {
        db *pgxpool.Pool
}

func NewSnippetRepository(db *pgxpool.Pool) *SnippetRepository {
        return &SnippetRepository{db: db}
}

// List returns every snippet, newest first. The id breaks the tie: created is
// identical for rows written in the same transaction, which the seed rows are.
func (r *SnippetRepository) List(ctx context.Context) ([]snippet.Snippet, error) {
        rows, err := r.db.Query(ctx, `
                SELECT id, title, content, created, expires
                FROM snippets
                ORDER BY created DESC, id DESC`)
        if err != nil {
                return nil, fmt.Errorf("query snippets: %w", err)
        }

        scanned, err := pgx.CollectRows(rows, pgx.RowToStructByPos[row])
        if err != nil {
                return nil, fmt.Errorf("collect snippets: %w", err)
        }

        // Allocated, not nil: a nil slice marshals to null and the contract says [].
        items := make([]snippet.Snippet, 0, len(scanned))
        for _, r := range scanned {
                items = append(items, r.toDomain())
        }

        return items, nil
}

// Get returns one snippet, or snippet.ErrNotFound.
func (r *SnippetRepository) Get(ctx context.Context, id int64) (snippet.Snippet, error) {
        rows, err := r.db.Query(ctx, `
                SELECT id, title, content, created, expires
                FROM snippets
                WHERE id = $1`, id)
        if err != nil {
                return snippet.Snippet{}, fmt.Errorf("query snippet %d: %w", id, err)
        }

        scanned, err := pgx.CollectExactlyOneRow(rows, pgx.RowToStructByPos[row])
        if errors.Is(err, pgx.ErrNoRows) {
                return snippet.Snippet{}, snippet.ErrNotFound
        }
        if err != nil {
                return snippet.Snippet{}, fmt.Errorf("collect snippet %d: %w", id, err)
        }

        return scanned.toDomain(), nil
}

// Insert writes one snippet and returns the row as the database stored it, so
// the caller never has to guess at created and expires.
func (r *SnippetRepository) Insert(ctx context.Context, in snippet.New) (snippet.Snippet, error) {
        rows, err := r.db.Query(ctx, `
                INSERT INTO snippets (title, content, created, expires)
                VALUES ($1, $2, now(), now() + make_interval(days => $3))
                RETURNING id, title, content, created, expires`, in.Title, in.Content, in.ExpiresInDays)
        if err != nil {
                return snippet.Snippet{}, fmt.Errorf("insert snippet: %w", err)
        }

        scanned, err := pgx.CollectExactlyOneRow(rows, pgx.RowToStructByPos[row])
        if err != nil {
                return snippet.Snippet{}, fmt.Errorf("collect inserted snippet: %w", err)
        }

        return scanned.toDomain(), nil
}

// Delete removes one snippet, or reports snippet.ErrNotFound when it matched
// nothing. The distinction matters: the second DELETE of one row is a 404.
func (r *SnippetRepository) Delete(ctx context.Context, id int64) error {
        tag, err := r.db.Exec(ctx, `DELETE FROM snippets WHERE id = $1`, id)
        if err != nil {
                return fmt.Errorf("delete snippet %d: %w", id, err)
        }
        if tag.RowsAffected() == 0 {
                return snippet.ErrNotFound
        }
        return nil
}

// Package service holds the rules: what a valid snippet is, and what the API
// does about one that is not. It knows nothing about HTTP and nothing about
// Postgres, which is what makes it testable without either.
package service

import (
        "context"
        "strings"
        "unicode/utf8"

        "github.com/saxeum/godough/internal/domain/snippet"
)

// SnippetRepository is the whole of what the service needs from storage. It is
// declared here, where it is consumed rather than where it is implemented, so
// the service can be exercised with a map and the Postgres package cannot leak a
// query detail upward.
type SnippetRepository interface {
        List(ctx context.Context) ([]snippet.Snippet, error)
        Get(ctx context.Context, id int64) (snippet.Snippet, error)
        Insert(ctx context.Context, in snippet.New) (snippet.Snippet, error)
        Delete(ctx context.Context, id int64) error
}

// SnippetService is the business layer for snippets.
type SnippetService struct {
        repo SnippetRepository
}

func NewSnippetService(repo SnippetRepository) *SnippetService {
        return &SnippetService{repo: repo}
}

// List returns every stored snippet, newest first.
func (s *SnippetService) List(ctx context.Context) ([]snippet.Snippet, error) {
        return s.repo.List(ctx)
}

// Get returns one snippet, or snippet.ErrNotFound.
func (s *SnippetService) Get(ctx context.Context, id int64) (snippet.Snippet, error) {
        if id < 1 {
                return snippet.Snippet{}, snippet.ErrNotFound
        }
        return s.repo.Get(ctx, id)
}

// Create validates the input and stores it, returning the row as the database
// wrote it.
func (s *SnippetService) Create(ctx context.Context, in snippet.New) (snippet.Snippet, error) {
        if bad := validate(in); len(bad) > 0 {
                return snippet.Snippet{}, &snippet.ValidationError{Fields: bad}
        }
        return s.repo.Insert(ctx, in)
}

// Delete removes one snippet, or reports snippet.ErrNotFound.
func (s *SnippetService) Delete(ctx context.Context, id int64) error {
        if id < 1 {
                return snippet.ErrNotFound
        }
        return s.repo.Delete(ctx, id)
}

// validate reports every bad field at once. The keys are the JSON names the
// client sent, because they are rendered under the matching input.
func validate(in snippet.New) map[string]string {
        bad := map[string]string{}

        switch {
        case strings.TrimSpace(in.Title) == "":
                bad["title"] = "is required"
        case utf8.RuneCountInString(in.Title) > snippet.TitleMaxRunes:
                // varchar(100) counts characters, not bytes.
                bad["title"] = "must be at most 100 characters"
        }

        if strings.TrimSpace(in.Content) == "" {
                bad["content"] = "is required"
        }

        if in.ExpiresInDays < snippet.ExpiryMinDays || in.ExpiresInDays > snippet.ExpiryMaxDays {
                bad["expiresInDays"] = "must be between 1 and 365"
        }

        return bad
}

package httpapi

import (
        "errors"
        "mime"
        "net/http"
)

// maxBodyBytes is the largest request body the application will read.
//
// It has to live here rather than in the contract: the generated strict handler
// decodes r.Body before its own StrictMiddlewareFunc chain runs, so a strict
// middleware is already too late. nginx happens to enforce the same number,
// which is why its absence was invisible until a request was sent to the port
// behind nginx.
//
// 1 MiB is what nginx was enforcing by default. #40 lowers it and reports an
// over-long `content` as a field error rather than a 413; this is the transport
// guard that makes that safe to do.
const maxBodyBytes = 1 << 20

// bodyGuards applies the two checks the generated code does not make: how large
// a body may be, and what media type it must claim. Both are request-shaping, so
// both belong in front of everything else.
func bodyGuards(next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                if r.Body != nil && r.Body != http.NoBody {
                        r.Body = http.MaxBytesReader(w, r.Body, maxBodyBytes)
                }

                // Only when there is a body to describe. A POST with nothing in it is not
                // a media-type problem, and routing must get the chance to answer 405
                // before anything here does: this runs in front of the mux, so a check
                // that fires on every POST would turn "wrong method" into "wrong type".
                // -1 means an unknown length (chunked), which still carries a body.
                if carriesBody(r.Method) && r.ContentLength != 0 && !isJSON(r.Header.Get("Content-Type")) {
                        writeJSON(w, http.StatusUnsupportedMediaType, errorBody(http.StatusUnsupportedMediaType))
                        return
                }

                next.ServeHTTP(w, r)
        })
}

func carriesBody(method string) bool {
        switch method {
        case http.MethodPost, http.MethodPut, http.MethodPatch:
                return true
        }
        return false
}

// isJSON tolerates the charset parameter real clients send. It is deliberately
// strict about the media type itself: text/plain, multipart/form-data and
// application/x-www-form-urlencoded are the three a cross-site <form> can send
// without a preflight, so accepting them is the first half of a CSRF problem
// that #55 exists to close.
func isJSON(contentType string) bool {
        if contentType == "" {
                return false
        }
        mediaType, _, err := mime.ParseMediaType(contentType)
        if err != nil {
                return false
        }
        return mediaType == "application/json"
}

// isBodyTooLarge reports whether the decoder failed because MaxBytesReader cut
// the body off, which is a 413 rather than the 400 a malformed body gets.
func isBodyTooLarge(err error) bool {
        var tooLarge *http.MaxBytesError
        return errors.As(err, &tooLarge)
}

package httpapi

import (
        "context"
        "log/slog"
        "net/http"
        "runtime/debug"
        "time"
)

// RequestTimeout bounds one request end to end, from the moment the router hands
// it over to the moment the handler returns.
//
// It must exceed the database's statement_timeout (postgres.StatementTimeout,
// 5s) so a slow query surfaces as a database error that names the cause. If the
// request deadline fired first, every slow query would look identical and the
// log would say only "deadline exceeded".
const RequestTimeout = 15 * time.Second

// recoverer turns a panic into the API's own 500 and keeps the process serving.
//
// Without it, a panic is caught by net/http, which closes the connection with no
// body and writes to http.Server.ErrorLog — a different stream in a different
// format from everything else the service logs.
func recoverer(log *slog.Logger, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                defer func() {
                        recovered := recover()
                        if recovered == nil {
                                return
                        }

                        // ErrAbortHandler is how a handler asks for the connection to be
                        // aborted, and net/http handles it itself. Swallowing it would keep
                        // a connection alive that the handler wanted gone.
                        if recovered == http.ErrAbortHandler {
                                panic(recovered)
                        }

                        log.Error("panic recovered",
                                "panic", recovered,
                                "stack", string(debug.Stack()),
                                "method", r.Method,
                                "uri", r.URL.RequestURI(),
                        )
                        writeJSON(w, http.StatusInternalServerError, errorBody(http.StatusInternalServerError))
                }()

                next.ServeHTTP(w, r)
        })
}

// requestDeadline puts a deadline on the request context, so a handler blocked on
// the database is cancelled by the server rather than by the client leaving.
//
// It is the context, not http.TimeoutHandler: pgx watches the context and stops
// the query, whereas a timeout handler would answer the client and leave the
// query running.
func requestDeadline(timeout time.Duration, next http.Handler) http.Handler {
        return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
                ctx, cancel := context.WithTimeout(r.Context(), timeout)
                defer cancel()

                next.ServeHTTP(w, r.WithContext(ctx))
        })
}

// Handler creates http.Handler with routing matching OpenAPI spec.
func Handler(si ServerInterface) http.Handler {
        return HandlerWithOptions(si, StdHTTPServerOptions{})
}

// ServeMux is an abstraction of [http.ServeMux].
type ServeMux interface {
        HandleFunc(pattern string, handler func(http.ResponseWriter, *http.Request))
        http.Handler
}

type StdHTTPServerOptions struct {
        BaseURL          string
        BaseRouter       ServeMux
        Middlewares      []MiddlewareFunc
        ErrorHandlerFunc func(w http.ResponseWriter, r *http.Request, err error)
}

// HandlerFromMux creates http.Handler with routing matching OpenAPI spec based on the provided mux.
func HandlerFromMux(si ServerInterface, m ServeMux) http.Handler {
        return HandlerWithOptions(si, StdHTTPServerOptions{
                BaseRouter: m,
        })
}

func HandlerFromMuxWithBaseURL(si ServerInterface, m ServeMux, baseURL string) http.Handler {
        return HandlerWithOptions(si, StdHTTPServerOptions{
                BaseURL:    baseURL,
                BaseRouter: m,
        })
}

// HandlerWithOptions creates http.Handler with additional options
func HandlerWithOptions(si ServerInterface, options StdHTTPServerOptions) http.Handler {
        m := options.BaseRouter

        if m == nil {
                m = http.NewServeMux()
        }
        if options.ErrorHandlerFunc == nil {
                options.ErrorHandlerFunc = func(w http.ResponseWriter, r *http.Request, err error) {
                        http.Error(w, err.Error(), http.StatusBadRequest)
                }
        }

        wrapper := ServerInterfaceWrapper{
                Handler:            si,
                HandlerMiddlewares: options.Middlewares,
                ErrorHandlerFunc:   options.ErrorHandlerFunc,
        }

        m.HandleFunc(http.MethodGet+" "+options.BaseURL+"/api/snippet", wrapper.ListSnippets)
        m.HandleFunc(http.MethodPost+" "+options.BaseURL+"/api/snippet", wrapper.CreateSnippet)
        m.HandleFunc(http.MethodDelete+" "+options.BaseURL+"/api/snippet/{id}", wrapper.DeleteSnippet)
        m.HandleFunc(http.MethodGet+" "+options.BaseURL+"/api/snippet/{id}", wrapper.GetSnippet)

        return m
}

// StrictServerInterface represents all server handlers.
type StrictServerInterface interface {
        // ListSnippets Every snippet, newest first.
        // (GET /api/snippet)
        ListSnippets(ctx context.Context, request ListSnippetsRequestObject) (ListSnippetsResponseObject, error)
        // CreateSnippet Create one snippet.
        // (POST /api/snippet)
        CreateSnippet(ctx context.Context, request CreateSnippetRequestObject) (CreateSnippetResponseObject, error)
        // DeleteSnippet Remove one snippet.
        // (DELETE /api/snippet/{id})
        DeleteSnippet(ctx context.Context, request DeleteSnippetRequestObject) (DeleteSnippetResponseObject, error)
        // GetSnippet One snippet.
        // (GET /api/snippet/{id})
        GetSnippet(ctx context.Context, request GetSnippetRequestObject) (GetSnippetResponseObject, error)
}

// StrictHandlerFunc and StrictMiddlewareFunc are type aliases rather than
// distinct named types, so middleware written against one generated package
// can be shared with every other generated package using the same framework.
type StrictHandlerFunc = func(ctx context.Context, w http.ResponseWriter, r *http.Request, request any) (any, error)
type StrictMiddlewareFunc = func(f StrictHandlerFunc, operationID string) StrictHandlerFunc

type StrictHTTPServerOptions struct {
        RequestErrorHandlerFunc  func(w http.ResponseWriter, r *http.Request, err error)
        ResponseErrorHandlerFunc func(w http.ResponseWriter, r *http.Request, err error)
}

func NewStrictHandler(ssi StrictServerInterface, middlewares []StrictMiddlewareFunc) ServerInterface {
        return &strictHandler{ssi: ssi, middlewares: middlewares, options: StrictHTTPServerOptions{
                RequestErrorHandlerFunc: func(w http.ResponseWriter, r *http.Request, err error) {
                        http.Error(w, err.Error(), http.StatusBadRequest)
                },
                ResponseErrorHandlerFunc: func(w http.ResponseWriter, r *http.Request, err error) {
                        http.Error(w, err.Error(), http.StatusInternalServerError)
                },
        }}
}

func NewStrictHandlerWithOptions(ssi StrictServerInterface, middlewares []StrictMiddlewareFunc, options StrictHTTPServerOptions) ServerInterface {
        if options.RequestErrorHandlerFunc == nil {
                options.RequestErrorHandlerFunc = func(w http.ResponseWriter, r *http.Request, err error) {
                        http.Error(w, err.Error(), http.StatusBadRequest)
                }
        }
        if options.ResponseErrorHandlerFunc == nil {
                options.ResponseErrorHandlerFunc = func(w http.ResponseWriter, r *http.Request, err error) {
                        http.Error(w, err.Error(), http.StatusInternalServerError)
                }
        }
        return &strictHandler{ssi: ssi, middlewares: middlewares, options: options}
}

// Package httpapi is the transport tier. It owns HTTP and nothing else: the
// contract in openapi.gen.go parses parameters and bodies, this package maps
// domain values to wire values and domain errors to documented status codes,
// and everything else is delegated to the service.
package httpapi

import (
        "encoding/json"
        "net/http"

        "github.com/saxeum/godough/internal/domain/snippet"
)

// toWire converts a stored snippet to the type the contract describes. It is a
// hand-written mapping on purpose: the generated type is the wire format, and
// keeping it separate from the domain means renaming a field in the spec is a
// change to this file and the spec, not to the business rules.
func toWire(in snippet.Snippet) Snippet {
        return Snippet{
                ID:      in.ID,
                Title:   in.Title,
                Content: in.Content,
                Created: in.Created,
                Expires: in.Expires,
        }
}

// toWireAll does the same for a list, allocating for the empty case so the
// response is [] rather than null.
func toWireAll(in []snippet.Snippet) []Snippet {
        out := make([]Snippet, 0, len(in))
        for _, item := range in {
                out = append(out, toWire(item))
        }
        return out
}

// errorBody is the shape every non-2xx response shares. It is the generated
// Error type, so a change to the contract reaches this function.
//
// 4xx bodies say what the status says, which is what a client would print
// anyway. A 5xx body says nothing useful on purpose: the cause is in the log,
// where it has a request context and is not shown to the caller.
func errorBody(status int) Error {
        if status >= 500 {
                return Error{Error: "internal server error"}
        }
        return Error{Error: http.StatusText(status)}
}

// writeJSON is the only way this package writes a non-generated body. Encoder,
// not Marshal, so the bytes match what the generated response visitors emit.
func writeJSON(w http.ResponseWriter, status int, body any) {
        w.Header().Set("Content-Type", "application/json")
        w.WriteHeader(status)
        _ = json.NewEncoder(w).Encode(body)
}

// Server implements StrictServerInterface, the generated interface that the
// contract defines. A documented status code that no branch produces does not
// compile, which is the reason to generate the interface rather than write the
// handlers by hand.
type Server struct {
        log      *slog.Logger
        snippets *service.SnippetService
        // ready is the readiness check. It is a function rather than a pool so this
        // package does not import a driver.
        ready func(ctx context.Context) error
}

func NewServer(log *slog.Logger, snippets *service.SnippetService, ready func(ctx context.Context) error) *Server {
        return &Server{log: log, snippets: snippets, ready: ready}
}

// Routes builds the mux. The generated handler is mounted on the same mux as the
// probes, so there is one server and one set of routes; the API's paths carry
// the /api prefix in the spec, so nothing is rewritten here.
func (s *Server) Routes() http.Handler {
        mux := http.NewServeMux()
        mux.HandleFunc("GET /healthz", s.healthz)
        mux.HandleFunc("GET /readyz", s.readyz)

        // Two callbacks, because the generated router fails in two places: a body
        // that will not decode (strict options) and a path or query parameter that
        // will not parse (the wrapper's own ErrorHandlerFunc). Both must answer in
        // the API's JSON shape, or a malformed id is the only text/plain response
        // the service ever sends.
        badRequest := func(w http.ResponseWriter, r *http.Request, err error) {
                // MaxBytesReader cutting the body off is a 413, not the 400 a malformed
                // body gets. Same callback, so the distinction has to be made here.
                if isBodyTooLarge(err) {
                        s.log.Warn("request body too large", "limit", maxBodyBytes, "method", r.Method, "uri", r.URL.RequestURI())
                        writeJSON(w, http.StatusRequestEntityTooLarge, errorBody(http.StatusRequestEntityTooLarge))
                        return
                }
                s.log.Warn("bad request", "err", err, "method", r.Method, "uri", r.URL.RequestURI())
                writeJSON(w, http.StatusBadRequest, errorBody(http.StatusBadRequest))
        }

        strict := NewStrictHandlerWithOptions(s, nil, StrictHTTPServerOptions{
                RequestErrorHandlerFunc: badRequest,
                // A handler returned an error: only unexpected failures reach here, since
                // the documented ones are returned as response objects.
                ResponseErrorHandlerFunc: func(w http.ResponseWriter, r *http.Request, err error) {
                        // A client that hung up is not a fault. Writing to its connection is
                        // pointless and logging it at ERROR makes the error rate measure
                        // browsers navigating away rather than anything going wrong.
                        if errors.Is(err, context.Canceled) {
                                s.log.Info("request abandoned by the client", "method", r.Method, "uri", r.URL.RequestURI())
                                return
                        }
                        // A request that ran out of its own deadline is the server's failure,
                        // not the client's, so it stays a 500 — with the distinction in the
                        // log, because "deadline exceeded" alone does not say which deadline.
                        if errors.Is(err, context.DeadlineExceeded) {
                                s.log.Error("request deadline exceeded", "err", err, "timeout", RequestTimeout, "method", r.Method, "uri", r.URL.RequestURI())
                                writeJSON(w, http.StatusInternalServerError, errorBody(http.StatusInternalServerError))
                                return
                        }
                        s.log.Error("server error", "err", err, "method", r.Method, "uri", r.URL.RequestURI())
                        writeJSON(w, http.StatusInternalServerError, errorBody(http.StatusInternalServerError))
                },
        })

        api := HandlerWithOptions(strict, StdHTTPServerOptions{
                BaseRouter:       mux,
                ErrorHandlerFunc: badRequest,
        })

        // Outside, not inside: the generated handler reads the body before any of its
        // own middleware sees the request. Outermost first, so a panic anywhere below
        // is recovered, and the deadline covers everything that follows.
        return recoverer(s.log, requestDeadline(RequestTimeout, bodyGuards(api)))
}

// healthz says the process is alive. It must not touch the database: a database
// blip is not a reason to restart the container.
func (s *Server) healthz(w http.ResponseWriter, _ *http.Request) {
        _, _ = w.Write([]byte("ok\n"))
}

// readyz says the server can serve real traffic, so it does touch the database.
func (s *Server) readyz(w http.ResponseWriter, r *http.Request) {
        ctx, cancel := context.WithTimeout(r.Context(), time.Second)
        defer cancel()

        if err := s.ready(ctx); err != nil {
                s.log.Error("readiness check failed", "err", err)
                writeJSON(w, http.StatusServiceUnavailable, errorBody(http.StatusServiceUnavailable))
                return
        }

        _, _ = w.Write([]byte("ok\n"))
}

// ListSnippets implements GET /api/snippet.
func (s *Server) ListSnippets(ctx context.Context, _ ListSnippetsRequestObject) (ListSnippetsResponseObject, error) {
        items, err := s.snippets.List(ctx)
        if err != nil {
                return nil, err
        }
        return ListSnippets200JSONResponse(toWireAll(items)), nil
}

// GetSnippet implements GET /api/snippet/{id}.
func (s *Server) GetSnippet(ctx context.Context, request GetSnippetRequestObject) (GetSnippetResponseObject, error) {
        item, err := s.snippets.Get(ctx, request.Id)
        switch {
        case errors.Is(err, snippet.ErrNotFound):
                return GetSnippet404JSONResponse{NotFoundJSONResponse: NotFoundJSONResponse(errorBody(http.StatusNotFound))}, nil
        case err != nil:
                return nil, err
        }
        return GetSnippet200JSONResponse(toWire(item)), nil
}

// CreateSnippet implements POST /api/snippet.
func (s *Server) CreateSnippet(ctx context.Context, request CreateSnippetRequestObject) (CreateSnippetResponseObject, error) {
        created, err := s.snippets.Create(ctx, snippet.New{
                Title:         request.Body.Title,
                Content:       request.Body.Content,
                ExpiresInDays: request.Body.ExpiresInDays,
        })

        var invalid *snippet.ValidationError
        switch {
        case errors.As(err, &invalid):
                return CreateSnippet422JSONResponse{
                        ValidationErrorJSONResponse: ValidationErrorJSONResponse{Errors: invalid.Fields},
                }, nil
        case err != nil:
                return nil, err
        }

        s.log.Info("snippet created", "id", created.ID)
        location := fmt.Sprintf("/api/snippet/%d", created.ID)
        return CreateSnippet201JSONResponse{
                Body:    toWire(created),
                Headers: CreateSnippet201ResponseHeaders{Location: &location},
        }, nil
}

// DeleteSnippet implements DELETE /api/snippet/{id}.
func (s *Server) DeleteSnippet(ctx context.Context, request DeleteSnippetRequestObject) (DeleteSnippetResponseObject, error) {
        err := s.snippets.Delete(ctx, request.Id)
        switch {
        case errors.Is(err, snippet.ErrNotFound):
                return DeleteSnippet404JSONResponse{NotFoundJSONResponse: NotFoundJSONResponse(errorBody(http.StatusNotFound))}, nil
        case err != nil:
                return nil, err
        }

        s.log.Info("snippet deleted", "id", request.Id)
        return DeleteSnippet204Response{}, nil
}
